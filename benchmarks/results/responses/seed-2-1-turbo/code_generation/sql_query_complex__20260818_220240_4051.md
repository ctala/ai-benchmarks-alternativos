### Query Final
```sql
WITH valid_orders AS (
    -- Paso 1: Filtramos solo órdenes COMPLETADAS de los ÚLTIMOS 6 MESES
    SELECT 
        id,
        customer_id,
        total
    FROM orders
    WHERE 
        status = 'completed'
        AND created_at >= CURRENT_DATE - INTERVAL '6 months'
),
customer_metrics AS (
    -- Paso 2: Calculamos métricas agregadas por cliente (solo con ≥3 órdenes)
    SELECT 
        c.id AS customer_id,
        c.name AS nombre,
        c.email AS email,
        c.country AS pais,
        COUNT(vo.id) AS total_orders,
        SUM(vo.total) AS total_revenue,
        AVG(vo.total) AS avg_order_value
    FROM customers c
    INNER JOIN valid_orders vo ON c.id = vo.customer_id
    GROUP BY c.id, c.name, c.email, c.country
    HAVING COUNT(vo.id) >= 3 -- Filtramos clientes con al menos 3 órdenes completadas
),
customer_categories AS (
    -- Paso 3: Calculamos totales por cliente y categoría de producto
    SELECT 
        vo.customer_id,
        p.category,
        SUM(oi.quantity) AS total_units, -- Cantidad total de unidades por categoría
        SUM(oi.quantity * oi.unit_price) AS category_revenue -- Revenue por categoría
    FROM valid_orders vo
    INNER JOIN order_items oi ON vo.id = oi.order_id
    INNER JOIN products p ON oi.product_id = p.id
    GROUP BY vo.customer_id, p.category
),
ranked_categories AS (
    -- Paso 4: Rankeamos categorías por cliente para quedarnos con la más comprada
    SELECT 
        customer_id,
        category AS categoria_mas_comprada,
        ROW_NUMBER() OVER (
            PARTITION BY customer_id 
            -- Criterio de "más comprada": mayor cantidad de unidades
            -- Desempate por nombre de categoría A-Z para evitar filas duplicadas
            ORDER BY total_units DESC, category ASC
        ) AS category_rank
    FROM customer_categories
)
-- Query final: unimos métricas del cliente con su categoría top
SELECT 
    cm.nombre,
    cm.email,
    cm.pais,
    cm.total_orders,
    cm.total_revenue,
    ROUND(cm.avg_order_value, 2) AS avg_order_value, -- Redondeamos a 2 decimales (moneda)
    rc.categoria_mas_comprada
FROM customer_metrics cm
INNER JOIN ranked_categories rc 
    ON cm.customer_id = rc.customer_id 
    AND rc.category_rank = 1 -- Nos quedamos solo con la categoría #1 de cada cliente
ORDER BY cm.total_revenue DESC
LIMIT 10; -- Top 10 clientes
```

#### 📌 Suposiciones clave
1. **Revenue total**: Se calcula como la suma del campo `total` de la tabla `orders` (monto final de cada orden, que puede incluir descuentos, impuestos o envíos).
2. **Categoría más comprada**: Se define como la categoría con mayor cantidad de unidades adquiridas por el cliente.
3. **Desempate en categorías**: Si un cliente tiene dos categorías con la misma cantidad de unidades, se elige la que tenga el nombre alfabéticamente primero para evitar filas duplicadas.
4. **Fecha de corte**: Se toman los últimos 6 meses calendario desde la fecha actual.

#### 3. Si quieres calcular el revenue solo de productos (sin descuentos/impuestos de la orden)
Reemplaza la CTE `customer_metrics` por esta versión (usa la suma de los items de la orden):
```sql
customer_metrics AS (
    SELECT 
        c.id AS customer_id,
        c.name AS nombre,
        c.email AS email,
        c.country AS pais,
        COUNT(DISTINCT vo.id) AS total_orders, -- Usamos DISTINCT para no contar items como órdenes
        SUM(oi.quantity * oi.unit_price) AS total_revenue,
        SUM(oi.quantity * oi.unit_price) / COUNT(DISTINCT vo.id) AS avg_order_value
    FROM customers c
    INNER JOIN valid_orders vo ON c.id = vo.customer_id
    INNER JOIN order_items oi ON vo.id = oi.order_id
    GROUP BY c.id, c.name, c.email, c.country
    HAVING COUNT(DISTINCT vo.id) >= 3
)
```

#### 4. Si quieres incluir empates en el top 10
Si hay clientes con el mismo `total_revenue` y quieres mostrar todos los que estén en el top 10 (incluso si son más de 10 filas), usa esta versión del query final:
```sql
WITH -- (mismas CTEs anteriores)
final_ranked AS (
    SELECT 
        cm.nombre,
        cm.email,
        cm.pais,
        cm.total_orders,
        cm.total_revenue,
        ROUND(cm.avg_order_value, 2) AS avg_order_value,
        rc.categoria_mas_comprada,
        RANK() OVER (ORDER BY cm.total_revenue DESC) AS overall_rank
    FROM customer_metrics cm
    INNER JOIN ranked_categories rc 
        ON cm.customer_id = rc.customer_id

SELECT *

ORDER BY overall_rank ASC;
```
